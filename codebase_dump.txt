import React, { useState } from 'react';
import AppHeader from './components/layout/AppHeader';
import Sidebar from './components/layout/Sidebar';
import StatusBar from './components/layout/StatusBar';
import TelemetryStage from './components/layout/TelemetryStage';
import { TelemetryProvider } from './context/TelemetryContext';
import { DEFAULT_CHANNELS, type TelemetryChannel } from './types/telemetry';
import styles from './App.module.scss';

const TelemetryStudioLayout: React.FC = () => {
  const [channels, setChannels] = useState<TelemetryChannel[]>(DEFAULT_CHANNELS);

  const handleToggleChannel = (id: string) => {
    setChannels((prev) =>
      prev.map((ch) => (ch.id === id ? { ...ch, enabled: !ch.enabled } : ch))
    );
  };

  return (
    <div className={styles.appContainer}>
      <div className={styles.headerWrapper}>
        <AppHeader />
      </div>
      <div className={styles.sidebarWrapper}>
        <Sidebar channels={channels} onToggleChannel={handleToggleChannel} />
      </div>
      <div className={styles.stageWrapper}>
        <TelemetryStage channels={channels} />
      </div>
      <div className={styles.footerWrapper}>
        <StatusBar />
      </div>
    </div>
  );
};

export const App: React.FC = () => {
  return (
    <TelemetryProvider>
      <TelemetryStudioLayout />
    </TelemetryProvider>
  );
};

export default App;

import React, { useEffect } from 'react';
import { createPortal } from 'react-dom';
import styles from './ModalPortal.module.scss';

export interface ModalPortalProps {
  children: React.ReactNode;
  isOpen: boolean;
  onClose: () => void;
  title?: string;
}

export const ModalPortal: React.FC<ModalPortalProps> = ({
  children,
  isOpen,
  onClose,
  title,
}) => {
  useEffect(() => {
    if (!isOpen) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        onClose();
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => {
      window.removeEventListener('keydown', handleKeyDown);
    };
  }, [isOpen, onClose]);

  if (!isOpen) {
    return null;
  }

  return createPortal(
    <div
      className={styles.overlay}
      onClick={onClose}
      role="presentation"
    >
      <div
        className={styles.modal}
        onClick={(e) => e.stopPropagation()}
        role="dialog"
        aria-modal="true"
        aria-label={title ?? 'Modal Window'}
      >
        {children}
      </div>
    </div>,
    document.body
  );
};

export default ModalPortal;

import { useTelemetry } from '../../context/TelemetryContext';
import ModalPortal from '../common/ModalPortal';
import styles from './AppHeader.module.scss';

export const AppHeader: React.FC = () => {
  const { parsedSession, loadFile, isLoading } = useTelemetry();
  const [isModalOpen, setIsModalOpen] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const session = parsedSession?.sessionInfo;
  const tagText = session
    ? `${session.carName} @ ${session.trackName}`
    : 'No Session Loaded';

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      loadFile(file);
      // Reset input value so re-selecting same file triggers change
      e.target.value = '';
    }
  };

  return (
    <header className={styles.header}>
      <div className={styles.brand}>
        <svg
          className={styles.logoIcon}
          viewBox="0 0 24 24"
          fill="none"
          stroke="currentColor"
          strokeWidth="2"

          strokeLinejoin="round"
          aria-hidden="true"

          <polyline points="22 12 18 12 15 21 9 3 6 12 2 12" />
        </svg>
        <h1 className={styles.title}>iRacing Telemetry Studio</h1>
      </div>

      <div className={styles.metaGroup}>
        <div className={styles.sessionTag}>{tagText}</div>

        <input
          type="file"
          ref={fileInputRef}
          accept=".ibt"
          style={{ display: 'none' }}
          onChange={handleFileChange}
          aria-label="Upload .ibt file"
        />

        <button
          type="button"
          className={styles.openBtn}
          onClick={() => fileInputRef.current?.click()}
          disabled={isLoading}
        >
          <svg
            width="14"
            height="14"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
            strokeLinecap="round"
            strokeLinejoin="round"

            <polyline points="17 8 12 3 7 8" />
            <line x1="12" y1="3" x2="12" y2="15" />
          </svg>
          {isLoading ? 'Loading...' : 'Open .ibt'}
        </button>

        <button
          type="button"
          className={styles.inspectBtn}
          onClick={() => setIsModalOpen(true)}
        >
          Inspect Session
        </button>
      </div>

      <ModalPortal
        isOpen={isModalOpen}
        onClose={() => setIsModalOpen(false)}
        title="Session Details"
      >
        <div className={styles.modalHeader}>
          <h3>Session Inspection</h3>

import React from 'react';
import { useTelemetry } from '../../context/TelemetryContext';
import { DEFAULT_CHANNELS, type TelemetryChannel } from '../../types/telemetry';
import styles from './Sidebar.module.scss';

export interface SidebarProps {
  channels?: TelemetryChannel[];
  onToggleChannel?: (id: string) => void;
}

export const Sidebar: React.FC<SidebarProps> = ({
  channels = DEFAULT_CHANNELS,
  onToggleChannel,
}) => {
  const {
    parsedSession,
    selectedBaseLap,
    selectedRefLap,
    selectLap,
  } = useTelemetry();

  const laps = parsedSession?.laps ?? [];

  return (
    <aside className={styles.sidebar}>
      <div className={styles.section}>
        <h2 className={styles.title}>Telemetry Channels</h2>
        <ul className={styles.channelList}>
          {channels.map((channel) => {
            const indicatorClass =
              channel.id in styles
                ? styles[channel.id as keyof typeof styles]
                : '';

            return (
              <li
                key={channel.id}
                className={`${styles.channelItem} ${!channel.enabled ? styles.disabled : ''}`}
                onClick={() => onToggleChannel?.(channel.id)}
              >
                <div className={styles.channelLabelGroup}>
                  <input
                    type="checkbox"
                    className={styles.checkbox}
                    checked={channel.enabled}
                    onChange={() => onToggleChannel?.(channel.id)}
                    onClick={(e) => e.stopPropagation()}
                    aria-label={`Toggle ${channel.label} channel`}
                  />
                  <span
                    className={`${styles.indicator} ${indicatorClass}`}
                    style={!indicatorClass ? { backgroundColor: channel.color } : undefined}
                    aria-hidden="true"
                  />
                  <span className={styles.channelName}>{channel.label}</span>
                </div>
                <span className={styles.channelUnit}>{channel.unit}</span>
              </li>
            );
          })}
        </ul>
      </div>

      <div className={styles.section}>
        <h2 className={styles.title}>Session Laps</h2>
        {laps.length === 0 ? (
          <div className={styles.emptyLaps}>No laps recorded. Load a telemetry file to view laps.</div>
        ) : (
          <ul className={styles.lapList}>
            {laps.map((lap) => {
              const isBase = selectedBaseLap === lap.lapNumber;
              const isRef = selectedRefLap === lap.lapNumber;
              const itemClass = `${styles.lapItem} ${isBase && isRef ? styles.isBoth : isBase ? styles.isBase : isRef ? styles.isRef : ''}`;

              return (
                <li key={lap.lapNumber} className={itemClass}>
                  <div className={styles.lapRow}>
                    <span className={styles.lapNumber}>Lap {lap.lapNumber}</span>
                    <span className={styles.lapTime}>{lap.lapTimeFormatted}</span>
                  </div>
                  <div className={styles.badgeGroup}>
                    <button
                      type="button"
                      className={`${styles.roleBtn} ${isBase ? styles.activeBase : ''}`}
                      onClick={() => selectLap(lap.lapNumber, 'base')}
                      title="Set as Base Lap"
                    >
                      {isBase ? '● BASE' : 'Base'}
                    </button>
                    <button
                      type="button"
                      className={`${styles.roleBtn} ${isRef ? styles.activeRef : ''}`}
                      onClick={() => selectLap(lap.lapNumber, 'reference')}
                      title="Toggle Reference Lap"
                    >
                      {isRef ? '● REF' : 'Ref'}
                    </button>
                  </div>
                </li>
              );
            })}
          </ul>
        )}
      </div>
    </aside>
  );
};

export default Sidebar;

import React from 'react';
import { useTelemetry } from '../../context/TelemetryContext';
import styles from './StatusBar.module.scss';

export const StatusBar: React.FC = () => {
  const { activeSample, cursorDistance, parsedSession } = useTelemetry();

  const cursorText = activeSample
    ? `${activeSample.distancePct.toFixed(2)}%`
    : `${cursorDistance.toFixed(2)}%`;

  const speedText = activeSample != null
    ? `${activeSample.speed.toFixed(1)} km/h`
    : '-- km/h';

  const throttleText = activeSample != null
    ? `${activeSample.throttle.toFixed(0)}%`
    : '--%';

  const brakeText = activeSample != null
    ? `${activeSample.brake.toFixed(0)}%`
    : '--%';

  const gearText = activeSample != null
    ? activeSample.gear === 0
      ? 'N'
      : activeSample.gear < 0
        ? 'R'
        : String(activeSample.gear)
    : '--';

  const steeringText = activeSample != null
    ? `${activeSample.steering.toFixed(1)}°`
    : '--°';

  return (
    <footer className={styles.statusBar}>
      <div className={styles.metricsGroup}>
        <div className={styles.metricItem}>
          <span className={styles.metricLabel}>Distance:</span>
          <span className={styles.metricValue}>{cursorText}</span>

import React, { useCallback, useEffect, useRef, useState } from 'react';
import { useTelemetry } from '../../context/TelemetryContext';

import CornerAnalysis from '../telemetry/CornerAnalysis';
import DeltaBand from '../telemetry/DeltaBand';
import TelemetryCanvas from '../telemetry/TelemetryCanvas';
import TrackMap from '../telemetry/TrackMap';
import styles from './TelemetryStage.module.scss';

export interface TelemetryStageProps {
  channels?: TelemetryChannel[];
}

interface SampleOption {
  id: string;
  label: string;
  path: string;
}

const SAMPLE_OPTIONS: SampleOption[] = [
  { id: 'real_openwheel', label: 'Real Open-Wheel Log', path: '/samples/real_openwheel.ibt' },
  { id: 'openwheel', label: 'Formula 3 @ Spa', path: 'openwheel' },
  { id: 'gt3', label: 'GT3 R @ Watkins Glen', path: 'gt3' },
];

export const TelemetryStage: React.FC<TelemetryStageProps> = ({
  channels = DEFAULT_CHANNELS,
}) => {
  const {
    parsedSession,
    selectedBaseLap,
    selectedRefLap,
    cursorDistance,
    setCursorDistance,
    loadFile,
    loadSample,
    isLoading,
    error,
  } = useTelemetry();

  const [isDragOver, setIsDragOver] = useState(false);
  const [isScrubbing, setIsScrubbing] = useState(false);
  const [isMapOpen, setIsMapOpen] = useState(true);
  const trackRef = useRef<HTMLDivElement>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  // Drag & drop handlers
  const handleDragOver = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setIsDragOver(true);
  };

  const handleDragLeave = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setIsDragOver(false);
  };

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setIsDragOver(false);

    const files = e.dataTransfer.files;
    if (files.length > 0) {
      loadFile(files[0]);
    }
  };

  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      loadFile(file);
      e.target.value = '';
    }
  };

  // Scrubber drag / click logic
  const updateCursorFromEvent = useCallback(
    (clientX: number) => {
      if (!trackRef.current) return;
      const rect = trackRef.current.getBoundingClientRect();
      const clickX = clientX - rect.left;
      const pct = Math.max(0, Math.min(100, (clickX / rect.width) * 100));
      setCursorDistance(pct);
    },
    [setCursorDistance]
  );

  const handleMouseDown = (e: React.MouseEvent) => {
    setIsScrubbing(true);
    updateCursorFromEvent(e.clientX);
  };

  useEffect(() => {
    if (!isScrubbing) return;

    const handleMouseMove = (e: MouseEvent) => {
      updateCursorFromEvent(e.clientX);
    };

    const handleMouseUp = () => {
      setIsScrubbing(false);
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseup', handleMouseUp);

    return () => {
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseup', handleMouseUp);
    };
  }, [isScrubbing, updateCursorFromEvent]);

  // Active base and reference lap
  const baseLap = parsedSession?.laps.find((l) => l.lapNumber === selectedBaseLap);
  const refLap = selectedRefLap != null
    ? parsedSession?.laps.find((l) => l.lapNumber === selectedRefLap)
    : null;

  const hasActiveSession = Boolean(parsedSession && baseLap);

  return (
    <main
      className={styles.stage}
      onDragOver={handleDragOver}
      onDragLeave={handleDragLeave}
      onDrop={handleDrop}
    >
      <input
        type="file"
        ref={fileInputRef}
        accept=".ibt"
        style={{ display: 'none' }}
        onChange={handleFileInputChange}
      />

      {!hasActiveSession ? (
        <div className={styles.gridArea}>
          <div
            className={`${styles.dropZone} ${isDragOver ? styles.dropZoneActive : ''}`}
            onClick={() => fileInputRef.current?.click()}
            role="region"
            aria-label="Telemetry file drop zone"
          >
            <div className={styles.iconWrapper}>
              <svg
                className={styles.icon}
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
                aria-hidden="true"
              >
                <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4" />
                <polyline points="17 8 12 3 7 8" />
                <line x1="12" y1="3" x2="12" y2="15" />
              </svg>
            </div>

            <h2 className={styles.title}>Drop iRacing Telemetry (.ibt) File Here</h2>
            <p className={styles.description}>
              Drag and drop any binary <code>.ibt</code> session telemetry file to parse
              high-frequency channel traces, synchronize laps, and analyze performance.
            </p>

            {error && <div className={styles.errorBanner}>{error}</div>}

            <div className={styles.actions} onClick={(e) => e.stopPropagation()}>

                {SAMPLE_OPTIONS.map((sample) => (
                  <button

                  >

                    >
                      <polygon points="5 3 19 12 5 21 5 3" />
                    </svg>
                    {sample.label}
                  </button>
                ))}
              </div>
              <span className={styles.browseHint}>or click anywhere in the box to browse files</span>
            </div>
          </div>
        </div>
      ) : (
        <div className={styles.plotArea}>
          {/* Floating top toolbar allowing quick switching between presets and toggling Track Map */}
          <div className={styles.floatingToolbar}>
            <span className={styles.toolbarLabel}>Presets:</span>
            {SAMPLE_OPTIONS.map((sample) => (
              <button
                key={sample.id}

                onClick={() => loadSample(sample.path)}
                disabled={isLoading}

import type { TelemetrySample } from '../../types/telemetry';
import styles from './CornerAnalysis.module.scss';

export interface CornerAnalysisProps {
  baseSamples: TelemetrySample[];
  refSamples: TelemetrySample[] | null;
  cursorPct: number;
}

interface CornerApex {
  index: number;
  apexDistPct: number;
  baseMinSpeed: number;
  brakeStartPct: number;
  peakBrakePct: number;
  exitThrottlePct: number;
}

export const CornerAnalysis: React.FC<CornerAnalysisProps> = ({
  baseSamples,
  refSamples,
  cursorPct,
}) => {
  // Detect corner apexes in base lap
  const corners: CornerApex[] = useMemo(() => {
    if (!baseSamples || baseSamples.length < 50) return [];

    const detected: CornerApex[] = [];
    const windowSize = 15; // window on either side to check local minimum
    let lastApexDist = -10;

    for (let i = windowSize; i < baseSamples.length - windowSize; i++) {
      const current = baseSamples[i];
      if (current.speed >= 175) continue; // Only consider speeds below 175 km/h for corner apexes
      if (current.distancePct - lastApexDist < 4.0) continue; // Min 4% distance between apexes

      // Check if current is local minimum in window
      let isMin = true;
      for (let j = i - windowSize; j <= i + windowSize; j++) {
        if (baseSamples[j].speed < current.speed) {
          isMin = false;
          break;
        }
      }

      if (isMin) {
        lastApexDist = current.distancePct;

        // Search backward for brake application point and peak brake
        let brakeStart = current.distancePct;
        let peakBrake = 0;
        for (let b = i; b >= Math.max(0, i - 120); b--) {
          const bs = baseSamples[b];
          if (bs.brake > peakBrake) peakBrake = bs.brake;
          if (bs.brake > 5) {
            brakeStart = bs.distancePct;
          } else if (bs.throttle > 50 && b < i - 15) {
            break;
          }
        }

        // Search forward for full throttle exit point
        let exitThrottle = current.distancePct;
        for (let f = i; f < Math.min(baseSamples.length, i + 120); f++) {
          const fs = baseSamples[f];
          if (fs.throttle >= 90) {
            exitThrottle = fs.distancePct;
            break;
          }
        }

        detected.push({
          index: detected.length + 1,
          apexDistPct: Number(current.distancePct.toFixed(1)),
          baseMinSpeed: Number(current.speed.toFixed(1)),
          brakeStartPct: Number(brakeStart.toFixed(1)),
          peakBrakePct: Math.round(peakBrake),
          exitThrottlePct: Number(exitThrottle.toFixed(1)),
        });
      }
    }

    return detected;
  }, [baseSamples]);

  // Find nearest corner to cursorPct
  const activeCorner = useMemo(() => {
    if (corners.length === 0) return null;
    let closest = corners[0];
    let minDiff = Math.abs(corners[0].apexDistPct - cursorPct);

    for (let i = 1; i < corners.length; i++) {
      const diff = Math.abs(corners[i].apexDistPct - cursorPct);
      if (diff < minDiff) {
        minDiff = diff;
        closest = corners[i];
      }
    }
    return closest;
  }, [corners, cursorPct]);

  // Find ref speed at the active corner's apex
  const refMinSpeed = useMemo(() => {
    if (!activeCorner || !refSamples || refSamples.length === 0) return null;
    let nearest = refSamples[0];
    let minDiff = Math.abs(nearest.distancePct - activeCorner.apexDistPct);

    for (let i = 1; i < refSamples.length; i++) {
      const diff = Math.abs(refSamples[i].distancePct - activeCorner.apexDistPct);
      if (diff < minDiff) {
        minDiff = diff;
        nearest = refSamples[i];
      }
    }
    return nearest.speed;
  }, [activeCorner, refSamples]);

  if (!activeCorner) {
    return (
      <div className={styles.cornerBar}>
        <span className={styles.zoneBadge}>Track Straightaway</span>
        <div className={styles.metricsPills}>
          <div className={styles.statPill}>
            <span className={styles.label}>Zone:</span>
            <span className={styles.value}>Full Throttle Segment</span>
          </div>
        </div>
      </div>
    );
  }

  const speedDelta = refMinSpeed != null ? activeCorner.baseMinSpeed - refMinSpeed : null;

  return (
    <div className={styles.cornerBar}>
      <div className={styles.zoneGroup}>
        <span className={styles.zoneBadge}>
          Turn {activeCorner.index} (Apex @ {activeCorner.apexDistPct}%)
        </span>
      </div>

      <div className={styles.metricsPills}>
        <div className={styles.statPill}>
          <span className={styles.label}>Min Speed:</span>
          <span className={styles.value}>{activeCorner.baseMinSpeed} km/h</span>
          {speedDelta != null && (
            <span
              className={speedDelta >= 0 ? styles.deltaPositive : styles.deltaNegative}
              title={`Delta vs Reference (${refMinSpeed?.toFixed(1)} km/h)`}
            >
              ({speedDelta >= 0 ? '+' : ''}
              {speedDelta.toFixed(1)} km/h)
            </span>
          )}
        </div>

        <div className={styles.statPill}>
          <span className={styles.label}>Brake Zone:</span>
          <span className={styles.value}>
            {activeCorner.brakeStartPct}% dist (Peak {activeCorner.peakBrakePct}%)
          </span>
        </div>

        <div className={styles.statPill}>
          <span className={styles.label}>Full Throttle Exit:</span>
          <span className={styles.value}>{activeCorner.exitThrottlePct}% dist</span>
        </div>
      </div>
    </div>
  );
};

export default CornerAnalysis;

import React, { useCallback, useEffect, useMemo, useRef } from 'react';
import type { TelemetrySample } from '../../types/telemetry';
import { calculateLapDelta } from '../../utils/deltaMath';
import styles from './DeltaBand.module.scss';

export interface DeltaBandProps {
  baseSamples: TelemetrySample[];
  refSamples: TelemetrySample[] | null;
  cursorPct: number;
  onCursorChange?: (pct: number) => void;
}

const PADDING = {
  left: 64,
  right: 24,
  top: 6,
  bottom: 6,
};

export const DeltaBand: React.FC<DeltaBandProps> = ({
  baseSamples,
  refSamples,
  cursorPct,
  onCursorChange,
}) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const isDraggingRef = useRef(false);

  const deltas = useMemo(() => {
    if (!refSamples || refSamples.length < 10) return [];
    return calculateLapDelta(baseSamples, refSamples);
  }, [baseSamples, refSamples]);

  // Find active delta value at cursorPct
  const activeDelta = useMemo(() => {
    if (deltas.length === 0) return null;
    const index = Math.max(0, Math.min(deltas.length - 1, Math.round(cursorPct / 0.2)));
    return deltas[index]?.deltaSeconds ?? 0;
  }, [deltas, cursorPct]);

  const draw = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const width = canvas.clientWidth;
    const height = canvas.clientHeight;
    if (width <= 0 || height <= 0) return;

    const dpr = window.devicePixelRatio || 1;
    if (canvas.width !== Math.floor(width * dpr) || canvas.height !== Math.floor(height * dpr)) {
      canvas.width = Math.floor(width * dpr);
      canvas.height = Math.floor(height * dpr);
    }

    ctx.save();
    ctx.scale(dpr, dpr);
    ctx.clearRect(0, 0, width, height);

    const plotWidth = Math.max(1, width - PADDING.left - PADDING.right);
    const plotHeight = Math.max(1, height - PADDING.top - PADDING.bottom);
    const midY = PADDING.top + plotHeight / 2;

    // Background panel
    ctx.fillStyle = '#121820';
    ctx.fillRect(PADDING.left, PADDING.top, plotWidth, plotHeight);
    ctx.strokeStyle = '#263244';
    ctx.lineWidth = 1;
    ctx.strokeRect(PADDING.left, PADDING.top, plotWidth, plotHeight);

    // Center Zero Line
    // High-contrast Center Zero Line
    ctx.strokeStyle = '#38bdf8';
    ctx.lineWidth = 1.5;
    ctx.setLineDash([4, 3]);
    ctx.globalAlpha = 0.9;
    ctx.beginPath();
    ctx.moveTo(PADDING.left, midY);
    ctx.lineTo(PADDING.left + plotWidth, midY);
    ctx.stroke();
    ctx.globalAlpha = 1;
    ctx.setLineDash([]);

    // Viewport title label
    ctx.font = '600 10px "JetBrains Mono", monospace';
    ctx.fillStyle = '#8b949e';
    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.fillText('LAP DELTA (Base vs Ref)', PADDING.left + 8, PADDING.top + 6);

    // Y Axis zero label
    ctx.font = '600 11px "JetBrains Mono", monospace';
    ctx.fillStyle = '#e6edf3';
    ctx.textAlign = 'right';
    ctx.textBaseline = 'middle';
    ctx.fillText('0.00s', PADDING.left - 8, midY);

    if (deltas.length > 0) {
      // Find max abs delta for dynamic vertical scaling (minimum range 0.5s)
      let maxAbs = 0.5;
      for (const d of deltas) {
        if (Math.abs(d.deltaSeconds) > maxAbs) {
          maxAbs = Math.abs(d.deltaSeconds);
        }
      }
      // Add slight headroom
      maxAbs = Math.min(10, maxAbs * 1.15);

      const getX = (distPct: number) =>
        PADDING.left + (Math.max(0, Math.min(100, distPct)) / 100) * plotWidth;

      const getY = (deltaSec: number) => {
        const clampedDelta = Math.max(-maxAbs, Math.min(maxAbs, deltaSec));
        // delta > 0 is base faster (top), delta < 0 is ref faster (bottom)
        return midY - (clampedDelta / maxAbs) * (plotHeight / 2);
      };

      // Draw rich filled segments with gradients
      ctx.save();
      ctx.beginPath();
      ctx.rect(PADDING.left, PADDING.top, plotWidth, plotHeight);
      ctx.clip();

      const greenGrad = ctx.createLinearGradient(0, midY, 0, PADDING.top);
      greenGrad.addColorStop(0, 'rgba(34, 197, 94, 0.25)');
      greenGrad.addColorStop(1, 'rgba(34, 197, 94, 0.75)');

      const redGrad = ctx.createLinearGradient(0, midY, 0, PADDING.top + plotHeight);
      redGrad.addColorStop(0, 'rgba(239, 68, 68, 0.25)');
      redGrad.addColorStop(1, 'rgba(239, 68, 68, 0.75)');

      for (let i = 0; i < deltas.length - 1; i++) {
        const d0 = deltas[i];
        const d1 = deltas[i + 1];

        const x0 = getX(d0.distancePct);
        const y0 = getY(d0.deltaSeconds);
        const x1 = getX(d1.distancePct);
        const y1 = getY(d1.deltaSeconds);

        const avgDelta = (d0.deltaSeconds + d1.deltaSeconds) / 2;

        ctx.fillStyle = avgDelta >= 0 ? greenGrad : redGrad;
        ctx.beginPath();
        ctx.moveTo(x0, midY);
        ctx.lineTo(x0, y0);
        ctx.lineTo(x1, y1);
        ctx.lineTo(x1, midY);
        ctx.closePath();
        ctx.fill();
      }

      // Draw delta outline curve with high contrast
      ctx.strokeStyle = '#ffffff';
      ctx.lineWidth = 2;
      ctx.beginPath();
      for (let i = 0; i < deltas.length; i++) {
        const x = getX(deltas[i].distancePct);
        const y = getY(deltas[i].deltaSeconds);
        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      }
      ctx.stroke();
      ctx.restore();

      // Top & Bottom scale limits with larger font size
      ctx.font = '600 11px "JetBrains Mono", monospace';
      ctx.fillStyle = '#22c55e';
      ctx.textAlign = 'right';
      ctx.textBaseline = 'top';
      ctx.fillText(`+${maxAbs.toFixed(1)}s`, PADDING.left - 8, PADDING.top + 4);

      ctx.fillStyle = '#ef4444';
      ctx.textBaseline = 'bottom';
      ctx.fillText(`-${maxAbs.toFixed(1)}s`, PADDING.left - 8, PADDING.top + plotHeight - 4);
    }

    // Vertical crosshair line matching main canvas
    const cursorX = PADDING.left + (Math.max(0, Math.min(100, cursorPct)) / 100) * plotWidth;
    ctx.save();
    ctx.strokeStyle = '#e6edf3';
    ctx.lineWidth = 1.25;
    ctx.setLineDash([3, 3]);
    ctx.beginPath();
    ctx.moveTo(cursorX, PADDING.top);
    ctx.lineTo(cursorX, PADDING.top + plotHeight);
    ctx.stroke();
    ctx.restore();

    ctx.restore();
  }, [deltas, cursorPct]);

  // Handle ResizeObserver
  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    const ro = new ResizeObserver(() => {
      draw();
    });
    ro.observe(container);

    draw();

    return () => {
      ro.disconnect();
    };
  }, [draw]);

  // Scrub handler
  const handlePointerCoord = useCallback(
    (clientX: number) => {
      if (!onCursorChange) return;
      const canvas = canvasRef.current;
      if (!canvas) return;
      const rect = canvas.getBoundingClientRect();
      const plotWidth = rect.width - PADDING.left - PADDING.right;
      if (plotWidth <= 0) return;

      const clickX = clientX - rect.left - PADDING.left;
      const pct = Math.max(0, Math.min(100, (clickX / plotWidth) * 100));
      onCursorChange(pct);
    },
    [onCursorChange]
  );

  const handlePointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
    isDraggingRef.current = true;
    (e.target as HTMLElement).setPointerCapture(e.pointerId);
    handlePointerCoord(e.clientX);
  };

  const handlePointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (isDraggingRef.current) {
      handlePointerCoord(e.clientX);
    }
  };

  const handlePointerUp = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (isDraggingRef.current) {
      isDraggingRef.current = false;
      try {
        (e.target as HTMLElement).releasePointerCapture(e.pointerId);
      } catch {
        // Ignored
      }
    }
  };

  const deltaText = activeDelta != null
    ? `${activeDelta >= 0 ? '+' : ''}${activeDelta.toFixed(3)}s`
    : null;

  const deltaStatusClass = activeDelta == null
    ? styles.neutral
    : activeDelta > 0.001
      ? styles.faster
      : activeDelta < -0.001
        ? styles.slower
        : styles.neutral;

  return (
    <div ref={containerRef} className={styles.deltaContainer}>
      <canvas
        ref={canvasRef}
        className={styles.canvas}
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onPointerCancel={handlePointerUp}
      />

      {(!refSamples || refSamples.length < 10) && (
        <div className={styles.emptyNotice}>
          <svg
            width="14"
            height="14"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
            strokeLinecap="round"
            strokeLinejoin="round"
            aria-hidden="true"
          >
            <circle cx="12" cy="12" r="10" />
            <line x1="12" y1="8" x2="12" y2="12" />
            <line x1="12" y1="16" x2="12.01" y2="16" />
          </svg>
          <span>Select a Reference Lap in the sidebar to compare deltas</span>
        </div>
      )}

      {deltaText && (
        <div className={`${styles.deltaBadge} ${deltaStatusClass}`}>
          <span>Δ {deltaText}</span>
        </div>
      )}
    </div>
  );
};

export default DeltaBand;

import type { TelemetryChannel, TelemetrySample } from '../../types/telemetry';

  bottom: 16,
  left: 64,
  right: 24,
  gap: 12,
};

interface ViewportDef {
  key: 'speed' | 'gear' | 'pedals' | 'steering';
  title: string;
  min: number;
  max: number;
  ticks: { val: number; label: string }[];
}

export const TelemetryCanvas: React.FC<TelemetryCanvasProps> = ({
  baseSamples,
  refSamples,
  cursorPct,
  onCursorChange,
  activeChannels,
}) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const isDraggingRef = useRef(false);

  // Active channel checks
  const isSpeedEnabled = activeChannels.find((c) => c.id === 'speed')?.enabled ?? true;
  const isGearEnabled = activeChannels.find((c) => c.id === 'gear')?.enabled ?? true;
  const isThrottleEnabled = activeChannels.find((c) => c.id === 'throttle')?.enabled ?? true;
  const isBrakeEnabled = activeChannels.find((c) => c.id === 'brake')?.enabled ?? true;
  const isSteeringEnabled = activeChannels.find((c) => c.id === 'steering')?.enabled ?? true;
  const isPedalsEnabled = isThrottleEnabled || isBrakeEnabled;

  // Render loop
  const draw = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const width = canvas.clientWidth;
    const height = canvas.clientHeight;
    if (width <= 0 || height <= 0) return;

    const dpr = window.devicePixelRatio || 1;
    if (canvas.width !== Math.floor(width * dpr) || canvas.height !== Math.floor(height * dpr)) {
      canvas.width = Math.floor(width * dpr);
      canvas.height = Math.floor(height * dpr);
    }

    ctx.save();
    ctx.scale(dpr, dpr);
    ctx.clearRect(0, 0, width, height);

    const plotWidth = Math.max(1, width - PADDING.left - PADDING.right);

    // Build active viewports dynamically based on enabled channels
    const activeVps: ViewportDef[] = [];
    if (isSpeedEnabled) {
      activeVps.push({
        key: 'speed',
        title: 'SPEED (km/h)',
        min: 0,
        max: 280,
        ticks: [
          { val: 0, label: '0' },
          { val: 50, label: '50' },
          { val: 100, label: '100' },
          { val: 150, label: '150' },
          { val: 200, label: '200' },
          { val: 250, label: '250' },
          { val: 280, label: '280' },
        ],
      });
    }
    if (isGearEnabled) {
      activeVps.push({
        key: 'gear',
        title: 'GEAR (Shift Trace)',
        min: 0,
        max: 7,
        ticks: [
          { val: 0, label: 'N' },
          { val: 1, label: '1' },
          { val: 2, label: '2' },
          { val: 3, label: '3' },
          { val: 4, label: '4' },
          { val: 5, label: '5' },
          { val: 6, label: '6' },
          { val: 7, label: '7' },
        ],

    const target = Math.max(0, Math.min(100, cursorPct));

      i++;
    }

    const p0 = trackPoints[i];
    const p1 = trackPoints[i + 1];
    const span = p1.distPct - p0.distPct;

    }

    const t = Math.max(0, Math.min(1, (target - p0.distPct) / span));
    return {

  if (!samples || samples.length === 0) return null;
  let closest = samples[0];
  let minDiff = Math.abs(closest.distancePct - pct);

  for (let i = 1; i < samples.length; i++) {
    const diff = Math.abs(samples[i].distancePct - pct);
    if (diff < minDiff) {
      minDiff = diff;
      closest = samples[i];
    }
  }
  return closest;
}

function findFastestLapNumber(laps: LapData[]): number | null {
  if (laps.length === 0) return null;
  const flyingLaps = laps.filter(
    (l) => l.lapNumber > 0 && l.samples.length >= 200 && l.lapTimeSeconds >= 15
  );
  const candidates = flyingLaps.length > 0 ? flyingLaps : laps;
  const fastest = candidates.reduce((best, curr) =>
    curr.lapTimeSeconds < best.lapTimeSeconds ? curr : best
  );
  return fastest.lapNumber;
}

  const [selectedRefLap, setSelectedRefLap] = useState<number | null>(null);

      if (baseLap && baseLap.samples.length > 0) {

          const parsed = parseIbt(arrayBuffer);
          applyParsedSession(parsed);

          );
        }

        applyParsedSession(parsed);
      } catch (err) {

        setError(message);
      } finally {

      }
    },
    [applyParsedSession]
  );

    <TelemetryContext.Provider
      value={{

        setCursorDistance,
      }}
    >

    </TelemetryContext.Provider>
  );
};

import type { LapData, ParsedSession, SessionInfo, TelemetrySample } from '../types/telemetry';

export interface VarHeader {
  type: number;
  offset: number;
  name: string;
}

export function formatLapTime(seconds: number): string {
  if (!seconds || seconds <= 0 || !Number.isFinite(seconds)) {
    return '--:--.---';
  }
  const mins = Math.floor(seconds / 60);
  const remainder = seconds % 60;
  const wholeSecs = Math.floor(remainder).toString().padStart(2, '0');
  const millis = Math.floor((remainder - Math.floor(remainder)) * 1000)
    .toString()
    .padStart(3, '0');
  return `${mins}:${wholeSecs}.${millis}`;
}

function clamp(val: number, min: number, max: number): number {
  return Math.min(Math.max(val, min), max);
}

function readCString(uint8: Uint8Array, start: number, maxLength: number): string {
  let end = start;
  const max = Math.min(start + maxLength, uint8.byteLength);
  while (end < max && uint8[end] !== 0) {
    end++;
  }
  return new TextDecoder('ascii').decode(uint8.subarray(start, end)).trim();
}

function extractYamlField(yaml: string, fieldName: string): string {
  const regex = new RegExp(
    `(?:^|\\n)\\s*${fieldName}\\s*:\\s*(?:'([^']*)'|"([^"]*)"|([^\\r\\n#]+))`,
    'm'
  );
  const match = yaml.match(regex);
  if (!match) return '';
  return (match[1] ?? match[2] ?? match[3] ?? '').trim();
}

function readVarValue(view: DataView, offset: number, type: number): number {
  switch (type) {
    case 1: // bool
      return view.getUint8(offset);
    case 2: // int32
      return view.getInt32(offset, true);
    case 3: // bitField (int32)
      return view.getUint32(offset, true);
    case 4: // float32
      return view.getFloat32(offset, true);
    case 5: // float64
      return view.getFloat64(offset, true);
    case 0: // char
      return view.getUint8(offset);
    default:
      return view.getFloat32(offset, true);
  }
}

export function parseIbt(buffer: ArrayBuffer): ParsedSession {
  if (buffer.byteLength < 64) {
    throw new Error('Invalid .ibt file: Buffer is too small to contain an iRacing header.');
  }

  const view = new DataView(buffer);
  const uint8 = new Uint8Array(buffer);

  // 1. Parse Header offsets
  const tickRate = view.getInt32(8, true) || 60;
  const sessionInfoLen = view.getInt32(16, true);
  const sessionInfoOffset = view.getInt32(20, true);
  const numVars = view.getInt32(24, true);
  const varHeaderOffset = view.getInt32(28, true);
  const bufLen = view.getInt32(36, true);

  // Strictly read record buffer start from byte 52
  const varBufOffset = view.getInt32(52, true);

  // 2. Extract Session YAML
  let sessionYaml = '';
  if (
    sessionInfoOffset > 0 &&
    sessionInfoLen > 0 &&
    sessionInfoOffset + sessionInfoLen <= buffer.byteLength
  ) {
    const yamlBytes = uint8.subarray(sessionInfoOffset, sessionInfoOffset + sessionInfoLen);
    sessionYaml = new TextDecoder('iso-8859-1').decode(yamlBytes);
  }

  const trackDisplayName = extractYamlField(sessionYaml, 'TrackDisplayName');
  const driverCarScreenName =
    extractYamlField(sessionYaml, 'DriverCarScreenName') ||
    extractYamlField(sessionYaml, 'CarScreenName');
  const driverCarEngName = extractYamlField(sessionYaml, 'DriverCarEngName');
  const driverSetupName = extractYamlField(sessionYaml, 'DriverSetupName');

  // Search session YAML for UserName or DriverHeadName. If not found or empty, return 'N/A'
  const rawUserName = extractYamlField(sessionYaml, 'UserName');
  const rawDriverHeadName = extractYamlField(sessionYaml, 'DriverHeadName');
  const userName =
    (rawUserName && rawUserName.trim()) ||
    (rawDriverHeadName && rawDriverHeadName.trim()) ||
    'N/A';

  // 3. Parse Variable Headers (144 bytes each)
  const varMap = new Map<string, VarHeader>();
  const VAR_HEADER_SIZE = 144;

  if (varHeaderOffset > 0 && numVars > 0) {
    for (let i = 0; i < numVars; i++) {
      const headerStart = varHeaderOffset + i * VAR_HEADER_SIZE;
      if (headerStart + VAR_HEADER_SIZE > buffer.byteLength) break;

      const type = view.getInt32(headerStart, true);
      const offset = view.getInt32(headerStart + 4, true);
      const name = readCString(uint8, headerStart + 16, 32);

      varMap.set(name, { type, offset, name });
    }
  }

  const speedVar = varMap.get('Speed');
  const throttleVar = varMap.get('Throttle');
  const brakeVar = varMap.get('Brake');
  const steeringVar = varMap.get('SteeringWheelAngle');
  const gearVar = varMap.get('Gear');
  const rpmVar = varMap.get('RPM');
  const lapVar = varMap.get('Lap');
  const lapDistPctVar = varMap.get('LapDistPct');
  const lapCurrentLapTimeVar = varMap.get('LapCurrentLapTime');

  // 4. Parse Record Data (Ensure varBufOffset > 0 and varBufOffset < buffer.byteLength)
  const samples: TelemetrySample[] = [];
  const effectiveBufLen = bufLen > 0 ? bufLen : 0;
  let totalTickCount = 0;

  if (
    effectiveBufLen > 0 &&
    varBufOffset > 0 &&
    varBufOffset < buffer.byteLength
  ) {
    for (
      let offset = varBufOffset;
      offset + effectiveBufLen <= buffer.byteLength;
      offset += effectiveBufLen
    ) {
      totalTickCount++;

      const rawLapDistPct = lapDistPctVar
        ? readVarValue(view, offset + lapDistPctVar.offset, lapDistPctVar.type)
        : 0;

      const rawSpeed = speedVar
        ? readVarValue(view, offset + speedVar.offset, speedVar.type)
        : 0;

      const rawThrottle = throttleVar
        ? readVarValue(view, offset + throttleVar.offset, throttleVar.type)
        : 0;

      const rawBrake = brakeVar
        ? readVarValue(view, offset + brakeVar.offset, brakeVar.type)
        : 0;

      const rawSteering = steeringVar
        ? readVarValue(view, offset + steeringVar.offset, steeringVar.type)
        : 0;

      const rawGear = gearVar
        ? readVarValue(view, offset + gearVar.offset, gearVar.type)
        : 0;

      const rawRPM = rpmVar
        ? readVarValue(view, offset + rpmVar.offset, rpmVar.type)
        : 0;

      const rawLap = lapVar
        ? readVarValue(view, offset + lapVar.offset, lapVar.type)
        : 0;

      const rawLapCurrentLapTime = lapCurrentLapTimeVar
        ? readVarValue(view, offset + lapCurrentLapTimeVar.offset, lapCurrentLapTimeVar.type)
        : 0;

      const sample: TelemetrySample = {
        distancePct: clamp(rawLapDistPct * 100, 0, 100),
        speed: rawSpeed * 3.6, // m/s to km/h
        throttle: rawThrottle * 100,
        brake: rawBrake * 100,
        steering: rawSteering * (180 / Math.PI), // radians to degrees
        gear: rawGear,
        rpm: rawRPM,
        lap: rawLap,
        lapTime: rawLapCurrentLapTime,
      };

      samples.push(sample);
    }
  }

  // 5. Group parsed samples into indexed laps
  const lapsMap = new Map<number, TelemetrySample[]>();
  for (const sample of samples) {
    const lapNumber = sample.lap ?? 0;
    let lapList = lapsMap.get(lapNumber);
    if (!lapList) {
      lapList = [];
      lapsMap.set(lapNumber, lapList);
    }
    lapList.push(sample);
  }

  // If only 1 single out-lap or ongoing lap exists without lap markers (rawLap === 0),
  // group all samples into Lap 1 with lapTimeSeconds = samples.length / tickRate
  if (samples.length > 0 && lapsMap.size === 1 && lapsMap.has(0)) {
    lapsMap.clear();
    lapsMap.set(1, samples);
  }

  const allLaps: LapData[] = [];
  for (const [lapNumber, lapSamples] of lapsMap.entries()) {
    const lastSampleTime = lapSamples[lapSamples.length - 1]?.lapTime ?? 0;
    const firstSampleTime = lapSamples[0]?.lapTime ?? 0;
    let lapTimeSeconds = lastSampleTime > firstSampleTime ? lastSampleTime - firstSampleTime : lastSampleTime;

    if (lapTimeSeconds <= 0) {
      lapTimeSeconds = lapSamples.length / tickRate;
    }

    allLaps.push({
      lapNumber,
      lapTimeSeconds,
      lapTimeFormatted: formatLapTime(lapTimeSeconds),
      samples: lapSamples,
    });
  }

  // Filter out laps with fewer than 50 samples or invalid time boundaries (< 5 seconds)
  let validLaps = allLaps.filter(
    (lap) => lap.samples.length >= 50 && lap.lapTimeSeconds >= 5
  );

  // Robust Lap Fallback:
  // If strict filtering leaves 0 laps, preserve all laps with >= 30 samples.
  if (validLaps.length === 0 && allLaps.length > 0) {
    validLaps = allLaps.filter((l) => l.samples.length >= 30);
  }

  // If still 0 laps but samples exist, group all samples into Lap 1
  if (validLaps.length === 0 && samples.length > 0) {
    const fallbackLapSeconds = samples.length / tickRate;
    validLaps = [
      {
        lapNumber: 1,
        lapTimeSeconds: fallbackLapSeconds,
        lapTimeFormatted: formatLapTime(fallbackLapSeconds),
        samples,
      },
    ];
  }

  // Sort laps by lapNumber
  validLaps.sort((a, b) => a.lapNumber - b.lapNumber);

  // Determine fastest lap (prioritizing full flying laps over short outlaps)
  let bestLapTimeFormatted = '--:--.---';
  if (validLaps.length > 0) {
    const flyingLaps = validLaps.filter(
      (l) => l.lapNumber > 0 && l.samples.length >= 200 && l.lapTimeSeconds >= 15
    );
    const candidateLaps = flyingLaps.length > 0 ? flyingLaps : validLaps;
    const fastest = candidateLaps.reduce((best, curr) =>
      curr.lapTimeSeconds < best.lapTimeSeconds ? curr : best
    );
    bestLapTimeFormatted = fastest.lapTimeFormatted;
  }

  const sessionInfo: SessionInfo = {
    trackName: trackDisplayName || 'N/A',
    carName: driverCarScreenName || driverCarEngName || 'N/A',
    driverName: userName,
    bestLapTime: bestLapTimeFormatted,
    setupName: driverSetupName || undefined,
    trackDisplayName: trackDisplayName || undefined,
    driverCarEngName: driverCarEngName || undefined,
  };

  return {
    sessionInfo,
    laps: validLaps,
    totalTickCount,
    tickRate,
  };
}

import type { TelemetrySample } from '../types/telemetry';

export interface LapDeltaPoint {
  distancePct: number;
  deltaSeconds: number;
}

function interpolateLapTime(
  samples: TelemetrySample[],
  targetDistPct: number,
  pointer: { idx: number }
): number {
  const len = samples.length;
  if (len === 0) return 0;
  if (targetDistPct <= samples[0].distancePct) return samples[0].lapTime;
  if (targetDistPct >= samples[len - 1].distancePct) return samples[len - 1].lapTime;

  let i = pointer.idx;
  // Ensure we don't start beyond bounds
  if (i >= len - 1) i = len - 2;

  // Advance pointer while next sample is still before target
  while (i < len - 2 && samples[i + 1].distancePct < targetDistPct) {
    i++;
  }
  pointer.idx = i;

  const s0 = samples[i];
  const s1 = samples[i + 1];
  const distSpan = s1.distancePct - s0.distancePct;

  if (distSpan <= 0.000001) {
    return s0.lapTime;
  }

  const t = Math.max(0, Math.min(1, (targetDistPct - s0.distancePct) / distSpan));
  return s0.lapTime + t * (s1.lapTime - s0.lapTime);
}

/**
 * Calculates time delta between base and reference laps at regular 0.2% distance intervals.
 * deltaSeconds = refLapTime - baseLapTime
 * (negative means ref lap is ahead, positive means base lap is ahead).
 */
export function calculateLapDelta(
  baseSamples: TelemetrySample[],
  refSamples: TelemetrySample[]
): LapDeltaPoint[] {
  if (!baseSamples || !refSamples || baseSamples.length < 10 || refSamples.length < 10) {
    return [];
  }

  // Ensure samples are sorted by distancePct
  const sortedBase = [...baseSamples].sort((a, b) => a.distancePct - b.distancePct);
  const sortedRef = [...refSamples].sort((a, b) => a.distancePct - b.distancePct);

  const basePointer = { idx: 0 };
  const refPointer = { idx: 0 };
  const results: LapDeltaPoint[] = [];

  const STEPS = 500; // 0% to 100% in 0.2% increments (501 points)

  for (let step = 0; step <= STEPS; step++) {
    const distPct = Number((step * 0.2).toFixed(2));
    const baseLapTime = interpolateLapTime(sortedBase, distPct, basePointer);
    const refLapTime = interpolateLapTime(sortedRef, distPct, refPointer);
    const deltaSeconds = refLapTime - baseLapTime;

    results.push({
      distancePct: distPct,
      deltaSeconds: Number(deltaSeconds.toFixed(4)),
    });
  }

  return results;
}

export function generateMockSession(type: MockPresetType): ParsedSession {
  const config = PRESETS[type];
  const tickRate = 60;
  const validLaps: LapData[] = [];
  let totalTickCount = 0;

  for (const lapSpec of config.laps) {
    const ticks = Math.round(lapSpec.duration * tickRate);
    totalTickCount += ticks;
    const samples: TelemetrySample[] = [];

    // Lap-specific variance (flying laps slightly faster in corners)
    const lapPaceFactor = lapSpec.duration / config.laps[config.laps.length - 1].duration;

    for (let t = 0; t < ticks; t++) {
      const distPct = (t / ticks) * 100;
      const lapTime = t / tickRate;

      // Track corner layout representation
      const cornerAngle = (distPct / 100) * Math.PI * 2 * config.cornersCount;
      const cornerInfluence = Math.sin(cornerAngle);
      const isCornerEntry = cornerInfluence < -0.3;
      const isCornerApex = cornerInfluence <= -0.7;
      const isCornerExit = cornerInfluence > 0.2;

      // Steering
      let steering = Math.sin((distPct / 100) * Math.PI * config.cornersCount) * 75;
      if (config.brakingProfile === 'snap') {
        // High-downforce formula car has sharper, higher-precision steering
        steering *= 1.15;
      }

      // Throttle & Brake based on vehicle dynamics
      let throttle: number;
      let brake: number;

      if (isCornerEntry) {
        throttle = 0;
        if (config.brakingProfile === 'snap') {
          // Open-wheel: immediate 100% threshold snap, fast bleed
          const entryProgress = (-cornerInfluence - 0.3) / 0.7;
          brake = Math.min(100, Math.max(0, entryProgress > 0.5 ? 98 : 45));
        } else if (config.brakingProfile === 'abs') {
          // GT3: heavy initial braking with ABS micro-flutter
          const absFlutter = 6 * Math.sin(t * 0.9);
          brake = Math.min(100, Math.max(0, 92 + absFlutter));
        } else {
          // MX-5: progressive smooth trail braking
          const trailProgress = (-cornerInfluence - 0.3) / 0.7;
          brake = Math.min(75, Math.max(0, trailProgress * 70));
        }
      } else if (isCornerApex) {
        brake = config.brakingProfile === 'trail' ? 15 : 0;
        throttle = config.brakingProfile === 'trail' ? 35 : 10;
      } else if (isCornerExit) {
        brake = 0;
        throttle = Math.min(100, Math.max(0, 50 + (cornerInfluence * 50)));
      } else {
        // Straightaway
        brake = 0;
        throttle = 100;
      }

      // Speed calculation
      let speed: number;
      if (isCornerEntry || isCornerApex) {
        const drop = (config.topSpeedKmh - config.minSpeedKmh) * (brake / 100);
        speed = Math.max(config.minSpeedKmh, config.topSpeedKmh - drop * lapPaceFactor);
      } else {
        const straightProgress = Math.max(0, (cornerInfluence + 1) / 2);
        speed = config.minSpeedKmh + (config.topSpeedKmh - config.minSpeedKmh) * straightProgress;
      }

      // Stepped Gear Shift Dynamics
      const [minGear, maxGear] = config.gears;
      let gear: number;
      if (config.brakingProfile === 'trail') {
        // MX-5 Lime Rock Park: gears 2 to 5
        if (speed < 105) gear = 2;
        else if (speed < 132) gear = 3;
        else if (speed < 158) gear = 4;
        else gear = 5;
      } else {
        // 6-speed gear stepping
        const gearStep = (config.topSpeedKmh - config.minSpeedKmh) / (maxGear - minGear + 1);
        gear = Math.min(
          maxGear,
          Math.max(minGear, Math.floor((speed - config.minSpeedKmh) / gearStep) + minGear)
        );
      }

      // RPM
      const rpm = 5500 + ((speed % config.gearRatio) / config.gearRatio) * 3500;

      samples.push({
        distancePct: Number(distPct.toFixed(3)),
        speed: Number(speed.toFixed(1)),
        throttle: Number(throttle.toFixed(1)),
        brake: Number(brake.toFixed(1)),
        steering: Number(steering.toFixed(1)),
        gear,
        rpm: Math.round(rpm),
        lap: lapSpec.lapNumber,
        lapTime: Number(lapTime.toFixed(3)),
      });
    }

    validLaps.push({
      lapNumber: lapSpec.lapNumber,
      lapTimeSeconds: lapSpec.duration,
      lapTimeFormatted: formatLapTime(lapSpec.duration),
      samples,
    });
  }

  // Find fastest lap
  const fastest = validLaps.reduce((best, curr) =>
    curr.lapTimeSeconds < best.lapTimeSeconds ? curr : best
  );

  const sessionInfo: SessionInfo = {
    trackName: config.trackName,
    carName: config.carName,
    driverName: config.driverName,
    bestLapTime: fastest.lapTimeFormatted,
    setupName: 'competition_baseline.sto',
    trackDisplayName: config.trackName,
  };

  return {
    sessionInfo,
    laps: validLaps,
    totalTickCount,
    tickRate,
  };
}

import type { TelemetrySample } from '../types/telemetry';

export interface TrackPoint {
  x: number;
  y: number;
  distPct: number;
}

/**
 * Generates a 2D track map layout from telemetry samples using numerical integration
 * of steering angle and vehicle velocity, with loop closure correction and normalization.
 */
export function generateTrackCoordinates(samples: TelemetrySample[]): TrackPoint[] {
  if (!samples || samples.length === 0) {
    return [];
  }

  const rawPoints: { x: number; y: number; distPct: number }[] = [];
  let heading = 0;
  let currentX = 0;
  let currentY = 0;

  for (let i = 0; i < samples.length; i++) {
    const s = samples[i];
    // Approximate angular change from steering angle and vehicle speed
    const deltaHeading = (s.steering * (Math.PI / 180) * 0.05) * (s.speed / 100);
    heading += deltaHeading;
    currentX += Math.cos(heading) * (s.speed / 3.6) * 0.016;
    currentY += Math.sin(heading) * (s.speed / 3.6) * 0.016;

    rawPoints.push({
      x: currentX,
      y: currentY,
      distPct: s.distancePct,
    });
  }

  const N = rawPoints.length;
  if (N <= 1) {
    return [{ x: 500, y: 500, distPct: samples[0]?.distancePct ?? 0 }];
  }

  // Loop closure correction:
  // Calculate drift between end and start, and distribute correction linearly across all points
  const startX = rawPoints[0].x;
  const startY = rawPoints[0].y;
  const endX = rawPoints[N - 1].x;
  const endY = rawPoints[N - 1].y;
  const driftX = endX - startX;
  const driftY = endY - startY;

  const closedPoints: { x: number; y: number; distPct: number }[] = [];
  let minX = Infinity;
  let maxX = -Infinity;
  let minY = Infinity;
  let maxY = -Infinity;

  for (let i = 0; i < N; i++) {
    const frac = i / (N - 1);
    const cx = rawPoints[i].x - frac * driftX;
    const cy = rawPoints[i].y - frac * driftY;

    if (cx < minX) minX = cx;
    if (cx > maxX) maxX = cx;
    if (cy < minY) minY = cy;
    if (cy > maxY) maxY = cy;

    closedPoints.push({
      x: cx,
      y: cy,
      distPct: rawPoints[i].distPct,
    });
  }

  // Normalize coordinates into a 0-1000 coordinate space with 40px padding
  const PADDING = 40;
  const TARGET_SIZE = 1000 - 2 * PADDING; // 920

  const spanX = maxX - minX || 1;
  const spanY = maxY - minY || 1;
  const scale = TARGET_SIZE / Math.max(spanX, spanY);

  const offsetX = PADDING + (TARGET_SIZE - spanX * scale) / 2;
  const offsetY = PADDING + (TARGET_SIZE - spanY * scale) / 2;

  return closedPoints.map((p) => ({
    x: Number((offsetX + (p.x - minX) * scale).toFixed(2)),
    y: Number((offsetY + (p.y - minY) * scale).toFixed(2)),
    distPct: p.distPct,
  }));
}
